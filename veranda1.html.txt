    function switchTab(tab) {
      const orderTab = document.getElementById('tab-order');
      const availTab = document.getElementById('tab-availability');
      const historyTab = document.getElementById('tab-history');

      const orderView = document.getElementById('view-order');
      const availView = document.getElementById('view-availability');
      const historyView = document.getElementById('view-history');
      
      const cartDrawer = document.getElementById('cart-drawer');

      // Сброс всех заголовочных вкладок
      [orderTab, availTab, historyTab].forEach(t => t.className = "flex-1 py-3 text-center font-medium text-gray-500 border-b-2 border-transparent");
      [orderView, availView, historyView].forEach(v => v.classList.add('hidden'));

      if (tab === 'order') {
        orderTab.className = "flex-1 py-3 text-center font-medium border-b-2 border-indigo-600 text-indigo-600";
        orderView.classList.remove('hidden');
        cartDrawer.classList.remove('hidden');
      } else if (tab === 'availability') {
        availTab.className = "flex-1 py-3 text-center font-medium border-b-2 border-indigo-600 text-indigo-600";
        availView.classList.remove('hidden');
        cartDrawer.classList.remove('hidden');
      } else if (tab === 'history') {
        historyTab.className = "flex-1 py-3 text-center font-medium border-b-2 border-indigo-600 text-indigo-600";
        historyView.classList.remove('hidden');
        cartDrawer.classList.add('hidden'); // Скрываем корзину на вкладке сохраненных заказов
      }
      render();
    }

    function renderCategories() {
      const categories = ['Все', ...new Set(menu.map(i => i.category))];
      
      const containerAvail = document.getElementById('categories-avail');
      containerAvail.innerHTML = categories.map(cat => `
        <button onclick="setCategoryAvail('${cat}')" 
          class="px-3 py-1.5 rounded-full text-xs font-semibold whitespace-nowrap transition ${
            activeCategoryAvail === cat 
              ? 'bg-indigo-600 text-white' 
              : 'bg-white text-gray-600 border border-gray-200'
          }">
          ${cat}
        </button>
      `).join('');

      const containerOrder = document.getElementById('categories-order');
      containerOrder.innerHTML = categories.map(cat => `
        <button onclick="setCategoryOrder('${cat}')" 
          class="px-3 py-1.5 rounded-full text-xs font-semibold whitespace-nowrap transition ${
            activeCategoryOrder === cat 
              ? 'bg-indigo-600 text-white' 
              : 'bg-white text-gray-600 border border-gray-200'
          }">
          ${cat}
        </button>
      `).join('');
    }

    function setCategoryAvail(cat) { activeCategoryAvail = cat; render(); }
    function setCategoryOrder(cat) { activeCategoryOrder = cat; render(); }

    function changeQuantity(id, delta) {
      const current = order[id] || 0;
      const next = current + delta;
      if (next <= 0) {
        delete order[id];
      } else {
        order[id] = next;
      }
      render();
    }

    function toggleAvailability(id, status) {
      const item = menu.find(i => i.id === id);
      if (item) {
        item.available = status;
        if (!status) delete order[id];
        saveMenu();
        render();
      }
    }

    function clearCart() {
      order = {};
      render();
    }

    // Управление модальным окном
    function openModal() {
      if (Object.keys(order).length === 0) return alert('Вы не выбрали ни одного блюда!');
      document.getElementById('order-modal').classList.remove('hidden');
    }

    function closeModal() {
      document.getElementById('order-modal').classList.add('hidden');
      document.getElementById('order-title').value = '';
      document.getElementById('order-note').value = '';
    }

    function confirmOrder() {
      const titleInput = document.getElementById('order-title').value.trim();
      const noteInput = document.getElementById('order-note').value.trim();

      const orderItems = Object.keys(order).map(id => {
        const item = menu.find(i => i.id == id);
        return {
          name: item.name,
          price: item.price,
          qty: order[id],
          sum: item.price * order[id]
        };
      });

      const totalSum = orderItems.reduce((acc, curr) => acc + curr.sum, 0);

      const newOrder = {
        id: Date.now(),
        title: titleInput || 'Без названия',
        note: noteInput,
        time: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }),
        items: orderItems,
        total: totalSum
      };

      savedOrders.unshift(newOrder);
      saveOrdersToStorage();

      order = {};
      closeModal();
      switchTab('history');
    }

    function deleteSavedOrder(orderId) {
      if (confirm('Удалить этот заказ?')) {
        savedOrders = savedOrders.filter(o => o.id !== orderId);
        saveOrdersToStorage();
        render();
      }
    }

          </div>
          <div class="flex gap-1.5 shrink-0">

            </button>
            <button onclick="toggleAvailability(${item.id}, true)" 
              class="px-3 py-2 rounded-xl font-bold text-xs transition ${item.available ? 'bg-emerald-500 text-white shadow-sm' : 'bg-gray-100 text-gray-400'}">
              + Есть
            </button>
          </div>
        </div>
      `).join('');

      // 2. Отрисовка Вкладки «Приём заказа»
      const availableMenu = menu.filter(i => i.available);
      const filteredOrderMenu = activeCategoryOrder === 'Все' 
        ? availableMenu 
        : availableMenu.filter(i => i.category === activeCategoryOrder);

      const orderList = document.getElementById('order-menu-list');

      if (availableMenu.length === 0) {
        orderList.innerHTML = `
          <div class="text-center py-10 px-4 bg-white rounded-2xl border border-dashed border-gray-300">
            <i class="fa-solid fa-utensils text-gray-300 text-3xl mb-2"></i>
            <p class="text-gray-500 font-medium text-sm">Нет блюд в наличии</p>
            <p class="text-gray-400 text-xs mt-1">Отметьте блюда во вкладке «1. Наличие»</p>
          </div>
        `;
      } else if (filteredOrderMenu.length === 0) {
        orderList.innerHTML = `
          <div class="text-center py-8 text-gray-400 text-xs">
            В категории «${activeCategoryOrder}» нет блюд в наличии
          </div>
        `;
      } else {
        orderList.innerHTML = filteredOrderMenu.map(item => {
          const qty = order[item.id] || 0;
          return `
            <div class="bg-white p-3 rounded-xl border border-gray-200 flex justify-between items-center shadow-sm">
              <div class="flex-1 pr-2">
                <div class="font-semibold text-gray-800 text-sm">${item.name}</div>
                <div class="text-xs text-gray-500">${item.price} ₽</div>
              </div>

              <div class="flex items-center gap-2 bg-gray-100 p-1 rounded-xl">
                <button onclick="changeQuantity(${item.id}, -1)" class="w-8 h-8 bg-white rounded-lg shadow-sm font-bold text-gray-700 active:bg-gray-200 flex items-center justify-center text-base">-</button>
                <span class="w-6 text-center font-bold text-sm ${qty > 0 ? 'text-indigo-600' : 'text-gray-400'}">${qty}</span>
                <button onclick="changeQuantity(${item.id}, 1)" class="w-8 h-8 bg-indigo-600 text-white rounded-lg shadow-sm font-bold active:bg-indigo-700 flex items-center justify-center text-base">+</button>
              </div>
            </div>
          `;
        }).join('');
      }

      // 3. Отрисовка Вкладки «Сохраненные Заказы»
      const historyContainer = document.getElementById('saved-orders-list');
      if (savedOrders.length === 0) {
        historyContainer.innerHTML = `
          <div class="text-center py-10 px-4 bg-white rounded-2xl border border-dashed border-gray-300">
            <i class="fa-solid fa-receipt text-gray-300 text-3xl mb-2"></i>
            <p class="text-gray-500 font-medium text-sm">Активных заказов нет</p>
            <p class="text-gray-400 text-xs mt-1">Оформить заказ можно во 2-й вкладке</p>
          </div>
        `;
      } else {
        historyContainer.innerHTML = savedOrders.map(o => `
          <div class="bg-white p-4 rounded-2xl border border-gray-200 shadow-sm space-y-3 relative">
            <div class="flex justify-between items-start border-b border-gray-100 pb-2">
              <div>
                <span class="font-bold text-indigo-600 text-base">${o.title}</span>
                <span class="text-gray-400 text-xs ml-2">${o.time}</span>
              </div>
              <button onclick="deleteSavedOrder(${o.id})" class="text-gray-400 hover:text-red-500 p-1">
                <i class="fa-solid fa-trash-can"></i>
              </button>
            </div>

            ${o.note ? `
              <div class="bg-amber-50 text-amber-800 text-xs p-2 rounded-lg border border-amber-200">
                <b>Заметка:</b> ${o.note}
              </div>
            ` : ''}

            <div class="divide-y divide-gray-50 text-xs">
              ${o.items.map(i => `
                <div class="flex justify-between py-1">
                  <span class="text-gray-700">${i.name} <b class="text-indigo-600">x${i.qty}</b></span>
                  <span class="font-semibold text-gray-900">${i.sum} ₽</span>
                </div>
              `).join('')}
            </div>

            <div class="flex justify-between items-center pt-2 border-t border-gray-100 font-bold text-sm">
              <span class="text-gray-600">Итого:</span>
              <span class="text-indigo-600 text-base">${o.total} ₽</span>
            </div>
          </div>
        `).join('');
      }

      // Обновление иконки счетчика активных заказов во вкладках
      const historyBadge = document.getElementById('history-count');
      if (savedOrders.length > 0) {
        historyBadge.innerText = savedOrders.length;
        historyBadge.classList.remove('hidden');
      } else {
        historyBadge.classList.add('hidden');
      }

      // 4. Обновление текущего черновика заказа (Нижняя панель)
      const summaryContainer = document.getElementById('cart-items-summary');
      const keys = Object.keys(order);
      let total = 0;
      let totalCount = 0;

      if (keys.length === 0) {
        summaryContainer.innerHTML = '<p class="text-gray-400 italic py-1">Заказ пуст</p>';
      } else {
        summaryContainer.innerHTML = keys.map(id => {
          const item = menu.find(i => i.id == id);
          if (!item) return '';
          const qty = order[id];
          const cost = item.price * qty;
          total += cost;
          totalCount += qty;
          return `
            <div class="flex justify-between items-center py-1.5">
              <span class="font-medium text-gray-700">${item.name} <b class="text-indigo-600">x${qty}</b></span>
              <span class="font-semibold text-gray-900">${cost} ₽</span>
            </div>
          `;
        }).join('');
      }

      document.getElementById('cart-total').innerText = `${total} ₽`;
      
      const badge = document.getElementById('order-badge');
      if (totalCount > 0) {
        badge.innerText = totalCount;
        badge.classList.remove('hidden');
      } else {
        badge.classList.add('hidden');
      }
    }

    render();
  </script>
</body>
</html>
